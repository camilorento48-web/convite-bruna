    <!-- FIREBASE INITIALIZATION & LOGIC -->

    <script type="module">
        import { initializeApp } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-app.js";
        import { getAuth, signInAnonymously, signInWithCustomToken, onAuthStateChanged } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-auth.js";
        import { getFirestore, collection, addDoc, onSnapshot, doc, deleteDoc } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-firestore.js";

        const appId = typeof __app_id !== 'undefined' ? __app_id : 'bruna-18-birthday-app';
        
        const defaultFirebaseConfig = {
            apiKey: "AIzaSyDemoKeyForBrunaBirthday18th",
            authDomain: "bruna-18-birthday.firebaseapp.com",
            projectId: "bruna-18-birthday",
            storageBucket: "bruna-18-birthday.appspot.com",
            messagingSenderId: "1234567890",
            appId: "1:1234567890:web:abc123def456"
        };

        const firebaseConfig = typeof __firebase_config !== 'undefined' ? JSON.parse(__firebase_config) : defaultFirebaseConfig;
        const initialAuthToken = typeof __initial_auth_token !== 'undefined' ? __initial_auth_token : null;

        let app, db, auth;
        let currentUser = null;
        let rsvpDataList = [];
        let isFirebaseAvailable = false;

        try {
            app = initializeApp(firebaseConfig);
            db = getFirestore(app);
            auth = getAuth(app);
            
            if (initialAuthToken) {
                signInWithCustomToken(auth, initialAuthToken).catch(err => {
                    signInAnonymously(auth);
                });
            } else {
                signInAnonymously(auth).catch(err => console.warn("Auth warning:", err));
            }

            onAuthStateChanged(auth, (user) => {
                if (user) {
                    currentUser = user;
                    isFirebaseAvailable = true;
                    listenToRSVPData();
                }
            });
        } catch (e) {
            console.warn("Local storage mode:", e);
            loadLocalStorageData();
        }

        // Realtime listener using strict path
        function listenToRSVPData() {
            if (!db || !currentUser) return;

            const rsvpCollectionRef = collection(db, 'artifacts', appId, 'public', 'data', 'rsvps');

            onSnapshot(rsvpCollectionRef, (snapshot) => {
                rsvpDataList = [];
                snapshot.forEach((docSnap) => {
                    rsvpDataList.push({
                        id: docSnap.id,
                        ...docSnap.data()
                    });
                });
                
                // Sort by time in memory
                rsvpDataList.sort((a, b) => (b.timestamp || 0) - (a.timestamp || 0));

                window.rsvpDataList = rsvpDataList;
                if (window.renderAdminTable) window.renderAdminTable();
            }, (error) => {
                console.error("Firestore Listener error:", error);
                loadLocalStorageData();
            });
        }

        function loadLocalStorageData() {
            const local = localStorage.getItem('bruna18_rsvp_backup');
            if (local) {
                try { rsvpDataList = JSON.parse(local); } catch(e) { rsvpDataList = []; }
            }
            window.rsvpDataList = rsvpDataList;
            if (window.renderAdminTable) window.renderAdminTable();
        }

        function saveLocalStorageData() {
            localStorage.setItem('bruna18_rsvp_backup', JSON.stringify(rsvpDataList));
        }

        window.saveRSVPEntry = async function(entry) {
            entry.timestamp = Date.now();
            entry.dateFormatted = new Date().toLocaleString('pt-BR', { dateStyle: 'short', timeStyle: 'short' });

            if (isFirebaseAvailable && db && currentUser) {
                try {
                    const rsvpCollectionRef = collection(db, 'artifacts', appId, 'public', 'data', 'rsvps');
                    await addDoc(rsvpCollectionRef, entry);
                    return true;
                } catch (err) {
                    console.error("Error saving to Firestore:", err);
                }
            }

            // Fallback
            rsvpDataList.unshift({ id: 'local_' + Date.now(), ...entry });
            saveLocalStorageData();
            window.rsvpDataList = rsvpDataList;
            if (window.renderAdminTable) window.renderAdminTable();
            return true;
        };

        window.deleteRSVPEntry = async function(id) {
            if (isFirebaseAvailable && db && currentUser && !id.startsWith('local_')) {
                try {
                    const docRef = doc(db, 'artifacts', appId, 'public', 'data', 'rsvps', id);
                    await deleteDoc(docRef);
                } catch (e) {
                    console.error("Delete error:", e);
                }
            }
            
            rsvpDataList = rsvpDataList.filter(item => item.id !== id);
            saveLocalStorageData();
            window.rsvpDataList = rsvpDataList;
            if (window.renderAdminTable) window.renderAdminTable();
            window.showToast("Registro removido!", "info");
        };
    </script>

        let currentFilter = 'all';
        let audioContext = null;
        let isMuted = false;

        function playFestiveSound() {
            if (isMuted) return;
            try {
                if (!audioContext) {
                    audioContext = new (window.AudioContext || window.webkitAudioContext)();
                }
                if (audioContext.state === 'suspended') {
                    audioContext.resume();
                }
                
                const notes = [523.25, 659.25, 783.99, 1046.50];
                notes.forEach((freq, idx) => {
                    const osc = audioContext.createOscillator();
                    const gain = audioContext.createGain();
                    osc.type = 'sine';
                    osc.frequency.value = freq;
                    gain.gain.setValueAtTime(0.12, audioContext.currentTime + idx * 0.08);
                    gain.gain.exponentialRampToValueAtTime(0.001, audioContext.currentTime + idx * 0.08 + 0.3);
                    osc.connect(gain);
                    gain.connect(audioContext.destination);
                    osc.start(audioContext.currentTime + idx * 0.08);
                    osc.stop(audioContext.currentTime + idx * 0.08 + 0.3);
                });
            } catch (e) {}
        }

        function toggleAudio() {
            isMuted = !isMuted;
            const icon = document.getElementById('soundIcon');
            if (isMuted) {
                icon.className = 'fa-solid fa-volume-xmark text-slate-400';
                showToast("Som desativado", "info");
            } else {
                icon.className = 'fa-solid fa-music text-pink-600';
                playFestiveSound();
                showToast("Efeitos sonoros ativados 🎶", "success");
            }
        }

        async function handleRsvpChoice(statusChoice) {
            const nameInput = document.getElementById('guestName');
            const statusDiv = document.getElementById('formStatus');

            const guestName = nameInput.value.trim();

            if (!guestName) {
                nameInput.focus();
                nameInput.classList.add('border-rose-500', 'ring-4', 'ring-rose-100');
                showToast("Por favor, digite seu nome e sobrenome!", "error");
                return;
            }
            nameInput.classList.remove('border-rose-500', 'ring-4', 'ring-rose-100');

            statusDiv.className = "text-center text-xs font-bold p-2.5 rounded-xl bg-pink-100 text-pink-700 block animate-pulse";
            statusDiv.innerHTML = `<i class="fa-solid fa-spinner fa-spin mr-1"></i> Registrando resposta...`;

            const entryData = {
                name: guestName,
                status: statusChoice,
            };

            const success = await window.saveRSVPEntry(entryData);

            if (success) {
                playFestiveSound();
                if (statusChoice === 'Estou confirmado.') {
                    triggerConfetti();
                }

                document.getElementById('rsvpForm').classList.add('hidden');
                const successCard = document.getElementById('rsvpSuccessCard');
                successCard.classList.remove('hidden');

                const successTitle = document.getElementById('successTitle');
                const successDesc = document.getElementById('successDesc');

                if (statusChoice === 'Estou confirmado.') {
                    successTitle.innerText = "Presença Confirmada! 🎉";
                    successDesc.innerText = `Prontinho, ${guestName}! Sua presença foi registrada para os 18 anos da Bruna.`;
                } else {
                    successTitle.innerText = "Resposta Registrada! 💌";
                    successDesc.innerText = `Obrigado por avisar, ${guestName}. A Bruna recebeu sua resposta.`;
                }

                showToast("Resposta registrada com sucesso!", "success");
            } else {
                statusDiv.className = "text-center text-xs font-bold p-2.5 rounded-xl bg-rose-100 text-rose-700 block";
                statusDiv.innerText = "Erro ao salvar. Tente novamente!";
            }
        }

        }

        function openAdminModal() {
            const modal = document.getElementById('adminAuthModal');
            modal.classList.remove('opacity-0', 'pointer-events-none');

        }

        function verifyAdminPassword() {
            const pwd = document.getElementById('adminPasswordInput').value.trim().toLowerCase();
            if (pwd === '18bruna' || pwd === 'bruna18' || pwd === '0201' || pwd === '18') {
                closeAdminAuthModal();
                openAdminDashboardModal();
                showToast("Acesso concedido ao Painel!", "success");
            } else {
                document.getElementById('authErrorMsg').classList.remove('hidden');
            }
        }

            modal.classList.remove('opacity-0', 'pointer-events-none');

        }

        function closeAdminDashboardModal() {

        function setFilter(filterType) {
            currentFilter = filterType;
            
            const btnAll = document.getElementById('filterAll');
            const btnConf = document.getElementById('filterConfirmed');
            const btnDecl = document.getElementById('filterDeclined');

            const activeClass = "px-3 py-1.5 text-xs font-bold rounded-lg bg-pink-600 text-white shadow-sm";
            const inactiveClass = "px-3 py-1.5 text-xs font-bold rounded-lg bg-white text-slate-600 hover:bg-slate-100 border border-slate-200";

            btnAll.className = filterType === 'all' ? activeClass : inactiveClass;
            btnConf.className = filterType === 'confirmed' ? activeClass : inactiveClass;
            btnDecl.className = filterType === 'declined' ? activeClass : inactiveClass;

            renderAdminTable();
        }

        window.renderAdminTable = function() {
            const data = window.rsvpDataList || [];
            const tbody = document.getElementById('adminTableBody');
            const emptyState = document.getElementById('emptyTableState');
            const searchVal = document.getElementById('adminSearchInput') ? document.getElementById('adminSearchInput').value.toLowerCase().trim() : '';

            const total = data.length;
            const confirmedCount = data.filter(item => item.status === 'Estou confirmado.').length;
            const declinedCount = data.filter(item => item.status === 'Não posso ir').length;

            document.getElementById('metricTotal').innerText = total;
            document.getElementById('metricConfirmed').innerText = confirmedCount;
            document.getElementById('metricDeclined').innerText = declinedCount;

            let filteredData = data.filter(item => {
                const matchesSearch = item.name.toLowerCase().includes(searchVal);
                if (!matchesSearch) return false;

                if (currentFilter === 'confirmed') return item.status === 'Estou confirmado.';
                if (currentFilter === 'declined') return item.status === 'Não posso ir';
                return true;
            });

            if (filteredData.length === 0) {
                tbody.innerHTML = '';
                emptyState.classList.remove('hidden');
                return;
            }

            emptyState.classList.add('hidden');
            
            tbody.innerHTML = filteredData.map(item => {
                const isConfirmed = item.status === 'Estou confirmado.';
                const badgeClass = isConfirmed 
                    ? 'bg-pink-100 text-pink-700 border-pink-300' 
                    : 'bg-slate-100 text-slate-600 border-slate-300';
                
                const statusIcon = isConfirmed ? 'fa-circle-check text-pink-600' : 'fa-circle-xmark text-slate-400';
                const dateDisplay = item.dateFormatted || new Date(item.timestamp || Date.now()).toLocaleString('pt-BR', { dateStyle: 'short', timeStyle: 'short' });

                return `
                    <tr class="hover:bg-pink-50/50 transition-colors">
                        <td class="py-3 px-4 font-bold text-slate-900">${escapeHtml(item.name)}</td>
                        <td class="py-3 px-4">
                            <span class="inline-flex items-center gap-1.5 px-2.5 py-1 rounded-full text-[11px] font-bold border ${badgeClass}">
                                <i class="fa-solid ${statusIcon}"></i>
                                ${escapeHtml(item.status)}
                            </span>
                        </td>
                        <td class="py-3 px-4 text-right text-slate-400 font-mono text-[11px]">${dateDisplay}</td>
                        <td class="py-3 px-4 text-center">
                            <button onclick="window.deleteRSVPEntry('${item.id}')" class="text-slate-400 hover:text-rose-600 p-1.5 rounded-lg hover:bg-rose-50 transition-colors" title="Excluir">
                                <i class="fa-solid fa-trash-can"></i>
                            </button>
                        </td>
                    </tr>
                `;
            }).join('');
        };

        function exportWhatsAppList() {
            const data = window.rsvpDataList || [];
            if (data.length === 0) {
                showToast("Não há confirmações para exportar!", "error");
                return;
            }

            const confirmedList = data.filter(i => i.status === 'Estou confirmado.');
            const declinedList = data.filter(i => i.status === 'Não posso ir');

            let text = `👑 *CONFIRMAÇÕES - BRUNA 18 ANOS* 🪩\n\n`;
            text += `✨ *Confirmados:* ${confirmedList.length}\n`;
            text += `❌ *Não poderão ir:* ${declinedList.length}\n\n`;
            
            text += `✅ *LISTA DE CONFIRMADOS (${confirmedList.length}):*\n`;
            if (confirmedList.length > 0) {
                confirmedList.forEach((item, idx) => {
                    text += `${idx + 1}. ${item.name}\n`;
                });
            } else {
                text += `Nenhum ainda.\n`;
            }

            if (declinedList.length > 0) {
                text += `\n❌ *NÃO PODERÃO IR (${declinedList.length}):*\n`;
                declinedList.forEach((item, idx) => {
                    text += `${idx + 1}. ${item.name}\n`;
                });
            }

            const textArea = document.createElement("textarea");
            textArea.value = text;
            document.body.appendChild(textArea);
            textArea.select();
            try {
                document.execCommand('copy');
                showToast("Lista copiada para o WhatsApp! 📋", "success");
            } catch (err) {
                showToast("Erro ao copiar lista", "error");
            }
            document.body.removeChild(textArea);
        }

        function triggerConfetti() {
            if (typeof confetti === 'function') {
                confetti({
                    particleCount: 70,
                    spread: 60,
                    origin: { y: 0.7 },
                    colors: ['#ec4899', '#f43f5e', '#fbbf24', '#ffffff']
                });
            }
        }

        function showToast(message, type = 'success') {
            const toast = document.getElementById('toast');
            const msgEl = document.getElementById('toastMsg');
            const iconEl = document.getElementById('toastIcon');

            msgEl.innerText = message;

            if (type === 'error') {
                iconEl.className = 'fa-solid fa-circle-exclamation text-rose-400 text-lg';
            } else if (type === 'info') {
                iconEl.className = 'fa-solid fa-circle-info text-sky-400 text-lg';
            } else {
                iconEl.className = 'fa-solid fa-circle-check text-emerald-400 text-lg';
            }

            toast.classList.remove('translate-y-20', 'opacity-0', 'pointer-events-none');
            
            setTimeout(() => {
                toast.classList.add('translate-y-20', 'opacity-0', 'pointer-events-none');
            }, 3500);
        }

        function escapeHtml(str) {
            if (!str) return '';
            return str.replace(/[&<>"']/g, function(m) {
                return {
                    '&': '&amp;',
                    '<': '&lt;',
                    '>': '&gt;',
                    '"': '&quot;',
                    "'": '&#039;'
                }[m];
            });
        }

        window.addEventListener('load', () => {
            const canvas = document.getElementById('sparkleCanvas');
            const ctx = canvas.getContext('2d');
            let width = canvas.width = window.innerWidth;
            let height = canvas.height = window.innerHeight;

            window.addEventListener('resize', () => {
                width = canvas.width = window.innerWidth;
                height = canvas.height = window.innerHeight;
            });

            const particles = [];
            for (let i = 0; i < 30; i++) {
                particles.push({
                    x: Math.random() * width,
                    y: Math.random() * height,
                    radius: Math.random() * 2 + 0.5,
                    alpha: Math.random(),
                    speed: Math.random() * 0.3 + 0.1,
                    color: ['#ec4899', '#fbbf24', '#ffffff', '#f472b6'][Math.floor(Math.random() * 4)]
                });
            }

            function animateSparkles() {
                ctx.clearRect(0, 0, width, height);
                particles.forEach(p => {
                    p.y -= p.speed;
                    p.alpha += (Math.random() - 0.5) * 0.04;

                    if (p.alpha < 0.1) p.alpha = 0.1;
                    if (p.alpha > 0.9) p.alpha = 0.9;
                    if (p.y < 0) { p.y = height; p.x = Math.random() * width; }

                    ctx.save();
                    ctx.globalAlpha = p.alpha;
                    ctx.fillStyle = p.color;
                    ctx.beginPath();
                    ctx.arc(p.x, p.y, p.radius, 0, Math.PI * 2);
                    ctx.fill();
                    ctx.restore();
                });
                requestAnimationFrame(animateSparkles);
            }
            animateSparkles();
        });
